class Node{
    public int data;
    public Node next;
    public Node(int date)
    {
        this.data=date;
      
    }
    public void  display()
    {
        System.out.println(data);
    }
}
class linkedlist{
    Node first;
    public void insertfirst(int data)
    {

        Node newnode=new Node(data);
        
        
            newnode.next=first;
            first=newnode;
       
    }
    public Node deletefirst()
    {
        Node temp=first;
        first=first.next;
        return temp;
    }
    public void displaylist()
    {
        Node current=first;
        while(current!=null)
        {
            current.display();
            current=current.next;
        }
    }
    public void insert_At_Tail(int value){
      Node newnode = new Node(value);
      if(first==null){
        first = newnode;
      }
      else{
        Node temp = first;
        while(temp.next!=null){
          temp = temp.next;
        }
        temp.next = newnode;
      }
    }

    public int find(int data)
{
    Node current=first;

    while (current!=null) {
        if(current.data==data)
        {
            return current.data;
        }
        current=current.next;
    }
    return -1;
   
}
public boolean deletedata(int data)
{
    if(first==null)
    {
        return false;
    }
    if(first.data==data)
    {
        first=first.next;
        return true;
    }
    Node current=first;
   while(current.next!=null)
   {
    if(current.next.data==data)
    {
        current.next=current.next.next;
        return true;
    }
    current=current.next;
   }
 return false;
}


}

class newstack{
int top;
int size;
int array[];

newstack(int size)
{
array=new int[size];
top=-1;
}

public boolean isempty()
{
    return (top==-1);
}
public boolean isfull()
{
    return(top==size-1);
}
public void push(int data)
{
    array[++top]=data;
}
public int pop()
{
    return array[top--];
    
}

public int peek()
{
    return array[top];
}

}
